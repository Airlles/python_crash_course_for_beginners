# 🔹 Hash Map Example

myMap = {}
myMap["alice"] = 90
myMap["bob"] = 70

print("alice" in myMap)   # True
print(myMap["alice"])     # 90

for key, val in myMap.items():
    print(key, val)
# alice 90
